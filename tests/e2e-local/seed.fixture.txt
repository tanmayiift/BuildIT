export const refreshAuth = internalMutation({ args: { userId: v.id("users") }, handler: async (ctx, args) => {
  const profile = await ctx.db.query("userProfiles").withIndex("by_user", q => q.eq("userId", args.userId)).unique();
  if (!profile) throw new Error("local_profile_missing");
  await ctx.db.patch(profile._id, { lastAuthenticatedAt: Date.now(), updatedAt: Date.now() });
} });
export const renewSession = internalMutation({ args: { userId: v.id("users"), sessionId: v.id("authSessions") }, handler: async (ctx, args) => {
  const session = await ctx.db.get(args.sessionId);
  if (!await ctx.db.get(args.userId) || (session && session.userId !== args.userId)) throw new Error("local_session_scope_invalid");
  // Signing out deliberately revokes the simulated session. Recreate it so another test run
  // can sign in again without re-seeding or overwriting either workspace's business fixtures.
  if (!session) return { sessionId: await ctx.db.insert("authSessions", { userId: args.userId, expirationTime: Date.now() + 8 * 3600000 }) };
  await ctx.db.patch(session._id, { expirationTime: Date.now() + 8 * 3600000 });
  return { sessionId: session._id };
} });
// Real Convex Auth refresh credentials for a simulated session. This function is copied only
// into the ignored, loopback-only backend, never into a production function directory.
export const browserSession = internalMutation({ args: { userId: v.id("users"), sessionId: v.id("authSessions") }, handler: async (ctx, args) => {
  const session = await ctx.db.get(args.sessionId);
  if (!session || session.userId !== args.userId || session.expirationTime < Date.now()) throw new Error("local_session_scope_invalid");
  const profile = await ctx.db.query("userProfiles").withIndex("by_user", q => q.eq("userId", args.userId)).unique();
  if (!profile) throw new Error("local_profile_missing");
  await ctx.db.patch(profile._id, { lastAuthenticatedAt: Date.now(), updatedAt: Date.now() });
  const tokenId = await ctx.db.insert("authRefreshTokens", { sessionId: args.sessionId, expirationTime: session.expirationTime });
  return { refreshToken: `${tokenId}|${args.sessionId}` };
} });
export const addMetric = internalMutation({ args: { organizationId: v.id("organizations"), reviewId: v.id("reviews") }, handler: async (ctx, args) => {
  const review = await ctx.db.get(args.reviewId);
  if (!review || review.organizationId !== args.organizationId) throw new Error("local_fixture_scope_invalid");
  await ctx.db.insert("metricEvents", { organizationId: args.organizationId, repositoryId: review.repositoryId, reviewId: review._id, name: "review_completed", value: 1, organizationTimezone: "UTC", occurredAt: Date.now() });
} });
